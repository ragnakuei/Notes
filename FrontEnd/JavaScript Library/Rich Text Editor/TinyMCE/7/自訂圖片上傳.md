# 自訂圖片上傳

## 在 paste 時，自訂圖片上傳邏輯

-   透過 images_upload_handler() 來自訂圖片上傳邏輯

```javascript
tinymce.init({
    license_key: 'gpl', // 免費授權
    selector: `#${this.id} #content`,
    language: 'zh_TW',

    resize: false,
    menubar: false,
    image_description: false,
    content_style: 'img { width: 20%;height: 20%; }',

    // statusbar:             false, // 隱藏下方狀態列
    // paste_data_images:     true,

    // 這個 method 會在以下的時機點被呼叫
    // 1. 以 paste 貼上內容，包含圖片時
    // 2. 有給定 images_upload_url 時，會在點選圖片上方的工具列時觸發
    images_upload_handler: (blobInfo, success, failure, progress) =>
        new Promise((resolve, reject) => {
            const blobInfoResult = {
                base64: blobInfo.base64(),
                blob: blobInfo.blob(),
                blobUri: blobInfo.blobUri(),
                filename: blobInfo.filename(),
                id: blobInfo.id(),
                name: blobInfo.name(),
                uri: blobInfo.uri(),
            };
            console.log('images_upload_handler > blobInfo', blobInfoResult);

            // 上傳圖片 url，該 api 要回傳此圖片的 url
            const uploadUrl = '/TinyMCE/UploadImage';
            const formData = new FormData();
            formData.append('id', this.querySelector('#id').value);
            formData.append('file', blobInfo.blob(), blobInfo.filename());

            fetch(uploadUrl, {
                method: 'POST',
                body: formData,
            })
                .then((response) => response.json())
                .then((data) => {
                    console.log('images_upload_handler > data', data);
                    resolve(data);
                    // resolve('https://picsum.photos/200/300')
                })
                .catch((error) => {
                    console.error('images_upload_handler > error', error);
                    reject(error);
                });
        }),

    // 讓 img src 可以以 / 開頭
    relative_urls: false,
    // setup:           function ( editor ) {
    // },
});
```
