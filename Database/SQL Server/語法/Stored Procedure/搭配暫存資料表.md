# 搭配暫存資料表


## INSERT INTO * EXEC

共用的 SP

```sql
CREATE PROCEDURE [dbo].[spA_fixed]
AS
BEGIN
	SET NOCOUNT ON;

	SELECT *
	FROM (
			 VALUES (1, 2),
					(1, NULL),
					(1, 1)
		 ) AS [TmpTable]([id], [status])

END
GO
```

執行語法

```sql
DECLARE @tmpTable TABLE (
    [id] INT,
    [status] INT
)

INSERT INTO @tmpTable
EXEC sp_fixed
```

### 缺點：無法進行第二次處理，會出現以下錯誤訊息 !

```sql
CREATE PROCEDURE [dbo].[spA_1]
AS
BEGIN
	SET NOCOUNT ON;

    DECLARE @tmpTable TABLE (
		[id] INT,
		[status] INT
	)

	INSERT INTO @tmpTable
	EXEC spA_fixed

END
GO
```


```
Msg 8164, Level 16, State 1, Procedure spA_1, Line 16 [Batch Start Line 0]
INSERT EXEC 陳述式不可以是巢狀的。
```

#### 解決方法 1

將查詢結果改為 OUTPUT & JSON

```sql
CREATE PROCEDURE [dbo].[spB_fixed]
    @Result NVARCHAR(MAX) OUTPUT
AS
BEGIN
    SET NOCOUNT ON;

    SELECT @Result = (
        SELECT [id], [status]
        FROM (
                VALUES (1, 2),
                       (1, NULL),
                       (1, 1)
            ) AS [TmpTable]([id], [status])
        FOR JSON PATH
    )
END
GO




CREATE PROCEDURE [dbo].[spB_1]
    @Result NVARCHAR(MAX) OUTPUT
AS 
BEGIN
    SET NOCOUNT ON;
    
    -- 直接將 spB_fixed 的結果傳給 @Result
    EXEC [dbo].[spB_fixed] @Result OUTPUT;
END
GO
```

以後以以下的語法執行，就不會出現錯誤訊息了 !


```sql
-- 執行 spB_fixed2 並將結果轉為 @TempTableJson
DECLARE @TempTableJson NVARCHAR(MAX);
EXEC [dbo].[spB_1] @Result = @TempTableJson OUTPUT;

-- 將 @TempTableJson 的內容插入到 @TempTable
DECLARE @TempTable TABLE (id INT, status INT);
INSERT INTO @TempTable (id, status)
SELECT id, status
FROM OPENJSON(@TempTableJson)
WITH (
    id INT,
    status INT
);


select *
from @TempTable
```
