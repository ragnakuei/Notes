- 也有用到 InMemoryChatMessageHistory


```python
import sys

from langchain_core.chat_history import BaseChatMessageHistory, InMemoryChatMessageHistory
from langchain_core.prompts import ChatPromptTemplate, MessagesPlaceholder
from langchain_core.runnables.history import RunnableWithMessageHistory

from langchainpractice.OpenAI import Initial_ChatOpenAI

# 初始化 OpenAI 接口 -> ChatOpenAI
llm = Initial_ChatOpenAI.llm

prompt = ChatPromptTemplate.from_messages([
    MessagesPlaceholder(variable_name="history"),
    ("human", "{user_input}"),
])

chain = prompt | llm

current_session_id = "test_session"
store = {}


def get_by_session_id(session_id: str) -> BaseChatMessageHistory:
    if session_id not in store:
        store[session_id] = InMemoryChatMessageHistory()
    return store[session_id]


chain_with_history = RunnableWithMessageHistory(
    chain,
    get_by_session_id,
    input_messages_key="user_input",
    history_messages_key="history",
)

while True:
    query = input(f"Prompt: ")

    if (query == "exit"
            or query == "quit"
            or query == "q"
            or query == "f"):
        break

    if query == '':
        continue

    response = chain_with_history.invoke(
        {"user_input": query},
        config={"configurable": {"session_id": current_session_id}}
    )
    print(response)

print('Exiting')
sys.exit()
```
