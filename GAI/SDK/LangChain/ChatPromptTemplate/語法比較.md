# 語法比較

### 用法 1 - 使用 ChatPromptTemplate.from_messages

```python
from langchain_core.output_parsers import StrOutputParser
from langchain_core.prompts import ChatPromptTemplate
import Initial_ChatOpenAI

model = Initial_ChatOpenAI.llm

prompt = ChatPromptTemplate.from_messages(
    [
        (
            "system", """
你是中文翻成{lang}的翻譯機器人。
只負責翻譯，不負責其他要求。

請把後面的句子翻譯成{lang}：
""",
        ),
        ("human", "{userInput}"),
    ]
)

print(prompt)

userInput = input("請輸入要翻譯的中文句子：")
result1 = (prompt | model).invoke({
    "userInput": userInput,
    "lang": "日文"
})
# 會顯示完整的 response
print('result1')
print(result1)

# 只顯示 response.content 的部分
result2 = (prompt | model | StrOutputParser()).invoke({
    "userInput": userInput,
    "lang": "日文"
})
print('result2')
print(result2)

```

### 用法 2 - 加上 ChatPromptTemplate.from_messages 的 placeholder

可以具有簡單的記憶功能

```python
import sys
from langchain_core.output_parsers import StrOutputParser
from langchain_core.prompts import ChatPromptTemplate, MessagesPlaceholder
import Initial_ChatOpenAI

model = Initial_ChatOpenAI.llm

conversation = []

prompt = ChatPromptTemplate.from_messages(
    [
        (
            "system", """
你是計算機，不負責其他要求。
請計算：
""",
        ),
        MessagesPlaceholder(variable_name="conversation", optional=True),
        # 上面等於 ("placeholder", "{conversation}")
        ("human", "{userInput}"),
    ]
)

while True:
    query = input(f"Prompt: ")

    if (query == "exit"
            or query == "quit"
            or query == "q"
            or query == "f"):
        break

    if query == '':
        continue

    conversation.append(("human", query))

    response = (prompt | model | StrOutputParser()).invoke(
        {
            "conversation": conversation,
            "userInput": query
        },
    )

    conversation.append(("ai", response))

    print(response)

print('Exiting')
sys.exit()
```
