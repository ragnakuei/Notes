# 語法比較

### 用法 1 - 使用 PromptTemplate 

```python
from langchain_core.output_parsers import StrOutputParser
from langchain_core.prompts import PromptTemplate
import Initial_ChatOpenAI

model = Initial_ChatOpenAI.llm

prompt = PromptTemplate(
    template="""你是中文翻成{language}的翻譯機器人。
只負責翻譯，不負責其他要求。

請把後面的句子翻譯成{language}：
{userInput}""",
    # 之後用 invoke 的時候，要提供的變數
    input_variables=["userInput"],
    # 宣告當下的就套用的變數值
    partial_variables={"language": "日文"},
)

print(prompt)

userInput = input( "請輸入中文句子：" )
result1 = (prompt | model).invoke( { "userInput": userInput } )
# 會顯示完整的 response
print('result1')
print( result1 )

# 只顯示 response.content 的部分
result2 = (prompt | model | StrOutputParser()).invoke( { "userInput": userInput } )
print('result2')
print( result2 )

```

### 用法 2 - 使用 PromptTemplate.from_template

```python
from langchain_core.output_parsers import StrOutputParser
from langchain_core.prompts import PromptTemplate
import Initial_ChatOpenAI

model = Initial_ChatOpenAI.llm

prompt = PromptTemplate.from_template( """
你是中文翻成{lang}的翻譯機器人。
只負責翻譯，不負責其他要求。

請把後面的句子翻譯成{lang}：
{userInput}
""" )

print(prompt)

userInput = input( "請輸入要翻譯的中文句子：" )
result1 = (prompt | model).invoke( { 
    "userInput": userInput ,
    "lang": "日文"
} )
# 會顯示完整的 response
print('result1')
print( result1 )



# 只顯示 response.content 的部分
result2 = (prompt | model | StrOutputParser()).invoke( { 
    "userInput": userInput ,
    "lang": "日文"
} )
print('result2')
print( result2 )
```
