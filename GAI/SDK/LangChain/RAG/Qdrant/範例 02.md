不使用 text splitter，手動將資料轉換為 documents，並且透過向量搜索找出最相似的問題後，將 問題 及 答案 交給 GAI 來整合回答

```python
import sys

from langchain.chains import create_retrieval_chain
from langchain.chains.combine_documents import create_stuff_documents_chain
from langchain_community.vectorstores import Qdrant
from langchain_core.documents import Document
from langchain_core.prompts import ChatPromptTemplate

import Initial_ChatOpenAI
from langchainpractice.OpenAI.ParseExcel import excelToDtos

dtos = excelToDtos('碳足跡QA.xlsx')
documents = [Document(f"問題:{dto.question} {dto.question} 答案：{dto.answer}") for dto in dtos]

model = Initial_ChatOpenAI.llm
embeddings_model = Initial_ChatOpenAI.embeddings

# 這邊需要套件 qdrant-client
qdrant = Qdrant.from_documents(
    documents,
    embeddings_model,
    # url="your qdrant cloud url",
    url="http://localhost:6333",
    # api_key="your key",
    collection_name="碳足跡QA",
    force_recreate=True,
)

retriever = qdrant.as_retriever()

prompt = ChatPromptTemplate.from_messages(
    [
        (
            "system", """
你是碳足跡 QA 機器人，只負責 碳足跡 QA 中的相關問題，不負責其他要求。

###
使用者詢問：{input}
###

###
相關的 context 如下：
{context}
###

請依照 context 裡的資訊來回答使用者的詢問：
""",
        )
    ]
)

while True:
    query = input(f"請發問: ")

    if (query == "exit"
            or query == "quit"
            or query == "q"
            or query == "f"):
        break

    if query == '':
        continue

    # create_stuff_documents_chain 會將 retriever 的資料放進 context 裡
    document_chain = create_stuff_documents_chain(model, prompt)
    retrieval_chain = create_retrieval_chain(retriever, document_chain)
    # 受套件限制，使用者輸入的部份必須是 "input"
    response = retrieval_chain.invoke({"input": query})

    print(f"詢問的問題是: {response['input']}")
    print("向量查詢的 context 是:")
    for doc in response['context']:
        print(f"_collection_name: {doc.metadata["_collection_name"]} page_content: {doc.page_content}")
    print(f"回答是: {response['answer']}\n\n")

print('Exiting')
sys.exit()

```
