# 使用者設定檔

路徑位於 C:\Users\{使用者名稱}\.claude\settings.json


### 拒絕讀取指定路徑的方式

```json
{
  "permissions": {
    "allow": [
        
    ],
    "deny": [
      "Read(//c/Windows/**)",
      "Edit(//c/Windows/**)",
      "Write(//c/Windows/**)",

      "Read(//c/Program Files/**)",
      "Edit(//c/Program Files/**)",
      "Write(//c/Program Files/**)",

      "Read(//c/Program Files (x86)/**)",
      "Edit(//c/Program Files (x86)/**)",
      "Write(//c/Program Files (x86)/**)",

      "Read(//c/ProgramData/**)",
      "Edit(//c/ProgramData/**)",
      "Write(//c/ProgramData/**)",

      "Read(~/Documents/**)",
      "Edit(~/Documents/**)",
      "Write(~/Documents/**)",

      "Read(~/Pictures/**)",
      "Edit(~/Pictures/**)",
      "Write(~/Pictures/**)",

      "Read(~/Videos/**)",
      "Edit(~/Videos/**)",
      "Write(~/Videos/**)",

      "Read(~/Desktop/**)",
      "Edit(~/Desktop/**)",
      "Write(~/Desktop/**)",

      "Read(~/Downloads/**)",
      "Edit(~/Downloads/**)",
      "Write(~/Downloads/**)",

      "Read(~/.ssh/**)",
      "Edit(~/.ssh/**)",
      "Write(~/.ssh/**)",

      "Read(~/.aws/**)",
      "Edit(~/.aws/**)",
      "Write(~/.aws/**)",

      "Read(~/.azure/**)",
      "Edit(~/.azure/**)",
      "Write(~/.azure/**)",

      "Read(~/AppData/**)",
      "Edit(~/AppData/**)",
      "Write(~/AppData/**)",

      "Read(**/.env)",
      "Edit(**/.env)",
      "Write(**/.env)",

      "Read(**/.env.*)",
      "Edit(**/.env.*)",
      "Write(**/.env.*)",

      "Read(**/*.pem)",
      "Edit(**/*.pem)",
      "Write(**/*.pem)",

      "Read(**/*.key)",
      "Edit(**/*.key)",
      "Write(**/*.key)",

      "Read(**/*.pfx)",
      "Edit(**/*.pfx)",
      "Write(**/*.pfx)",

      "Read(**/*.p12)",
      "Edit(**/*.p12)",
      "Write(**/*.p12)"
    ]
  }
}
```
