# 驗証碼圖片

```cs
/// <summary>
/// 驗証碼 商業邏輯元件
/// </summary>
/// <remarks>
/// 以 Windows Form 的元件實作，不能將站台放在 Linux 主機上 !
/// </remarks>
public class ValidationCodeService
{
    public ValidationCodeDto NewValidationCode()
    {
        var result = new ValidationCodeDto();

        // 驗證碼字數
        result.Code = RandomCode(4);

        // result.Image = NormalValidationCodeImage(result);
        result.Image = StandaloneValidationCodeImage(result);

        return result;
    }

    /// <summary>
    /// 每個字元都依照標準位置/角度繪製
    /// </summary>
    private string NormalValidationCodeImage(ValidationCodeDto result)
    {
        var ms = new MemoryStream();

        using (var map = new Bitmap(90, 25))
        {
            // 畫筆,在指定畫板畫板上畫圖
            using (var g = Graphics.FromImage(map))
            {
                g.Clear(Color.White);
                g.DrawString(result.Code, new Font("黑體", 19.0F), Brushes.Blue, new Point(10, -5));

                // 繪製干擾線(數字代表幾條)
                PaintInterLine(g,
                               10,
                               map.Width,
                               map.Height,
                               Brushes.Red);

                // PaintInterLine(g, 10, map.Width, map.Height, Brushes.Green);

                // 繪制干擾點
                // PaintPoints(g, 30, map.Width, map.Height, Brushes.Red);
                // PaintPoints(g, 30, map.Width, map.Height, Brushes.Green);
            }

            map.Save(ms, ImageFormat.Jpeg);
        }

        var image = ms.GetBuffer();
        return Convert.ToBase64String(image);
    }

    private string RandomCode(int length)
    {
        var s    = "0123456789abcdefghijklmnopqrstuvwxyz";
        var sb   = new StringBuilder();
        var rand = new Random();
        int index;

        for (var i = 0; i < length; i++)
        {
            index = rand.Next(0, s.Length);
            sb.Append(s[index]);
        }

        return sb.ToString();
    }

    private void PaintPoints(Graphics graphics,
                             int      pointCount,
                             int      mapWidth,
                             int      mapHeight,
                             Brush    brushColor)
    {
        var r = new Random();
        int x, y;

        for (var i = 0; i < pointCount; i++)
        {
            x = r.Next(0, mapWidth);
            y = r.Next(0, mapHeight);
            graphics.DrawRectangle(new Pen(brushColor),
                                   x,
                                   y,
                                   1,
                                   1);
        }
    }

    private void PaintInterLine(Graphics g,
                                int      lineCount,
                                int      width,
                                int      height,
                                Brush    brushColor)
    {
        var r = new Random();
        int startX, startY, endX, endY;

        for (var i = 0; i < lineCount; i++)
        {
            startX = r.Next(0, width);
            startY = r.Next(0, height);
            endX   = r.Next(0, width);
            endY   = r.Next(0, height);
            g.DrawLine(new Pen(brushColor),
                       startX,
                       startY,
                       endX,
                       endY);
        }
    }

    /// <summary>
    /// 針對每個字元，繪製獨立圖片，再組合
    /// </summary>
    private string StandaloneValidationCodeImage(ValidationCodeDto result)
    {
        var resultImgSizeX = 92;
        var resultImgSizeY = 37;
        var imgSizeX       = resultImgSizeX / result.Code.Length;

        var offsetX = 0;
        var offsetY = 0;

        // offset 亂數正負範圍
        var offsetRange = 3;

        var rotateAngle = 0;

        // rotate 亂數正負範圍
        var rotateRange = 20;

        var rand = new Random();

        var charImages = result.Code
                               .Select(c =>
                                {
                                    var map = new Bitmap(imgSizeX, resultImgSizeY);

                                    using (var g = Graphics.FromImage(map))
                                    {
                                        g.Clear(Color.White);

                                        rotateAngle = rand.Next(-rotateRange, rotateRange);
                                        offsetX     = rand.Next(-offsetRange, offsetRange);
                                        offsetY     = rand.Next(-offsetRange, offsetRange);

                                        g.TranslateTransform(offsetX, offsetY);
                                        g.RotateTransform(rotateAngle);
                                        g.DrawString(c.ToString(), new Font("微軟正黑體", 19.0F), Brushes.Blue, new Point(0, -5));
                                    }

                                    return map;
                                })
                               .ToArray();
        var resultImg = MergeImages(charImages);

        var ms = new MemoryStream();
        resultImg.Save(ms, ImageFormat.Jpeg);
        return Convert.ToBase64String(ms.GetBuffer());
    }

    private Image MergeImages(Bitmap[] images)
    {
        try
        {
            // 确定所有图像的总宽度和最大高度
            int totalWidth = images.Sum(img => img.Width);
            int maxHeight  = images[0].Height;

            // 创建一个新的图像
            var mergedImage = new Bitmap(totalWidth, maxHeight);

            using (Graphics g = Graphics.FromImage(mergedImage))
            {
                // 将每个图像绘制到合并图像上
                int currentWidth = 0;
                foreach (var img in images)
                {
                    g.DrawImage(img, currentWidth, 0);
                    currentWidth += img.Width;
                    img.Dispose();
                }

                // 繪製干擾線(數字代表幾條)
                PaintInterLine(g,
                               10,
                               mergedImage.Width,
                               mergedImage.Height,
                               Brushes.Red);

                // PaintInterLine(g, 10, map.Width, map.Height, Brushes.Green);

                // 繪制干擾點
                // PaintPoints(g, 30, map.Width, map.Height, Brushes.Red);
                // PaintPoints(g, 30, map.Width, map.Height, Brushes.Green);
            }

            return mergedImage;
        }
        catch (Exception e)
        {
            throw;
        }
    }
}
```
