# 與第三方元件互動

### 驗証呼叫次數

```cs
[Test]
public async Task StartAsync_SpecifyScheduleTasks_ReceivedCalled()
{
    // Arrange
    var sampleTaskMock = new Mock<IScheduleTaskService>();

    var scheduleTasksProviderMock = new Mock<IScheduleTasksProvider>();
    scheduleTasksProviderMock.Setup(x => x.GetTasks()).Returns(new[] { sampleTaskMock.Object });

    var target = new App(new Mock<ILogger<App>>().Object,
                         new Mock<IHostApplicationLifetime>().Object,
                         scheduleTasksProviderMock.Object,
                         new Mock<IOptionsMonitor<AppSettingDto>>().Object);

    // Act
    await target.StartAsync(default);

    // Assert
    sampleTaskMock.Verify(x => x.RunAsync(It.IsAny<ScheduleTaskContextDto>(), default), Times.Once);
}
```

### 依照指定參數回傳給定資料

```cs
[Test]
public async Task RunAsync_所有Org都有資料_ReturnSpecifyContent()
{
    // Arrange
    var store = new Dictionary<string, PublicParsingDto[]>
                {
                    ["56"] = new PublicParsingDto[]
                                {
                                    new()
                                    {
                                        ColumnA    = "ColumnA01",
                                        ColumnB = "ColumnB01",
                                        ColumnC = "ColumnC01",
                                        ColumnD   = "ColumnD01",
                                    },
                                    new()
                                    {
                                        ColumnA    = "ColumnA02",
                                        ColumnB = "ColumnB02",
                                        ColumnC = "ColumnC02",
                                        ColumnD   = "ColumnD02",
                                    },
                                },
                    ["57"] = new PublicParsingDto[]
                                {
                                    new()
                                    {
                                        ColumnA    = "ColumnA03",
                                        ColumnB = "ColumnB03",
                                        ColumnC = "ColumnC03",
                                        ColumnD   = "ColumnD03",
                                    },
                                },
                    ["27"] = new PublicParsingDto[]
                                {
                                    new()
                                    {
                                        ColumnA    = "ColumnA06",
                                        ColumnB = "ColumnB06",
                                        ColumnC = "ColumnC06",
                                        ColumnD   = "ColumnD06",
                                    },
                                    new()
                                    {
                                        ColumnA    = "ColumnA07",
                                        ColumnB = "ColumnB07",
                                        ColumnC = "ColumnC07",
                                        ColumnD   = "ColumnD07",
                                    },
                                }
                };

    var actualDtos = new Dictionary<string, PublicParsingDto[]>();

    var parsingRepositoryMock = new Mock<IParsingRepository>();
    foreach (var orgcd in store.Keys)
    {
        parsingRepositoryMock.Setup(x => x.PublicParsingAsync(2023, orgcd))
                                .Callback<int, string>((year, orgcd) =>
                                                    {
                                                        actualDtos.Add(orgcd, Array.Empty<PublicParsingDto>());
                                                    })
                                .ReturnsAsync(store[orgcd]);
    }

    var sendMailServiceMock = new Mock<ISendMailService>();
    var actualContextDtos   = new Dictionary<string, ScheduleTaskContextDto>();
    sendMailServiceMock.Setup(x => x.PublicParsing(It.IsAny<string>(),
                                                    It.IsAny<ScheduleTaskContextDto>(),
                                                    It.IsAny<PublicParsingDto[]>()))
                        .Callback<string, ScheduleTaskContextDto, PublicParsingDto[]>((orgcd, context, dtos) =>
                                                                                        {
                                                                                            actualContextDtos[orgcd] = context;
                                                                                            actualDtos[orgcd]        = dtos;
                                                                                        });

    var target = new PublicParsingService(parsingRepositoryMock.Object,
                                            sendMailServiceMock.Object);

    var scheduleTaskContextDto = new ScheduleTaskContextDto
                                    {
                                        AppSettingDto = new()
                                                        {
                                                            RunDateTime = new DateTime(2023, 2, 3),
                                                        }
                                    };

    // Act
    await target.RunAsync(scheduleTaskContextDto, CancellationToken.None);

    // Assert
    actualContextDtos.Should().HaveCount(5);
    actualDtos.Should().HaveCount(5);

    foreach (var orgcd in store.Keys)
    {
        actualContextDtos.Should().ContainKey(orgcd);
        actualContextDtos[orgcd].Should().BeEquivalentTo(scheduleTaskContextDto);

        actualDtos.Should().ContainKey(orgcd);
        actualDtos[orgcd].Should().BeEquivalentTo(store[orgcd]);
    }
}
```
