# 產生實體的幾個方式

### Method Signature

```cs
using System.Linq.Expressions;
using System.Reflection.Emit;

public class Factory(string dependency)
{
    public IService Create<T>(bool isAsync)
        where T : IService
    {
        // TODO
    }
}
```

### Pattern Matching

方便 Trace Code

```cs
var type = typeof(T).Name;

return type switch
       {
           nameof(AService) => new AService(dependency, isAsync),
           nameof(BService) => new BService(dependency, isAsync),
           _                         => throw new ArgumentException(message: "invalid type", paramName: nameof(type))
       };
```

### Activator

```cs
var instance = Activator.CreateInstance(typeof(T), dependency, isAsync);
return (IService)instance;
```

### Method Invoke

```cs
var type = typeof(T);
var constructor = type.GetConstructor(new[] {typeof(string), typeof(bool)});
if (constructor == null)
{
    throw new ArgumentException(message: "invalid type", paramName: nameof(type));
}

var instance = constructor.Invoke(new object[] {dependency, isAsync});
return (Service) instance;
```

### Expression

```cs
var type = typeof(T);
var instance = Expression.New(type.GetConstructor(new[] { typeof(string), typeof(bool) }),
                              Expression.Constant(dependency),
                              Expression.Constant(isAsync));
if (instance == null)
{
    throw new ArgumentException(message: "invalid type", paramName: nameof(type));
}

return Expression.Lambda<Func<Service>>(instance).Compile()();
```

### Emit

```cs
var type        = typeof(T);
var constructor = type.GetConstructor(new[] { typeof(string), typeof(bool) });
if (constructor == null)
{
    throw new ArgumentException(message: "invalid type", paramName: nameof(type));
}

var method = new DynamicMethod("CreateInstance", type, new[] { typeof(string), typeof(bool) }, type);
var il     = method.GetILGenerator();
il.Emit(OpCodes.Ldarg_0);
il.Emit(OpCodes.Ldarg_1);
il.Emit(OpCodes.Newobj, constructor);

il.Emit(OpCodes.Ret);
return ((Func<string, bool, T>)method.CreateDelegate(typeof(Func<string, bool, T>)))(dependency, isAsync);
```
